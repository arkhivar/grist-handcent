name: SMS → Grist sync

on:
  repository_dispatch:
    types: [sms-sync]
  workflow_dispatch:
    inputs:
      cid:
        description: "Conversation: 276 / 277 / 0321 / 900 / all"
        required: false
        default: "all"
        type: string
  # Safety net while the phone-side repository_dispatch trigger is being
  # tested. This schedule burns Actions minutes (~500–900 min/month); DELETE
  # this block once dispatch proves stable.
  schedule:
    - cron: "*/15 * * * *"

concurrency:
  group: sms-sync
  cancel-in-progress: false

jobs:
  sync:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install deps
        run: pip install -r requirements.txt

      - name: Run sync
        env:
          # dispatch payload cid (fallback all) → workflow input → all
          SYNC_CID: ${{ github.event_name == 'repository_dispatch' && (github.event.client_payload.cid || 'all') || (github.event_name == 'workflow_dispatch' && inputs.cid || 'all') }}
          HANDCENT_AUTH: ${{ secrets.HANDCENT_AUTH }}
          GRIST_API_KEY: ${{ secrets.GRIST_API_KEY }}
        run: python runner.py | tee artifact.json
