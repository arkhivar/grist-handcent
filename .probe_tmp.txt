on:
  repository_dispatch:
    types: [sms-sync]
  schedule:
    - cron: "*/15 * * * *"

jobs:
  sync:
    runs-on: ubuntu-latest
    steps:
      - name: Run sync
        env:
          HANDCENT_AUTH: ${{ secrets.HANDCENT_AUTH }}
        run: python runner.py | tee artifact.json
      - name: Upload artifact
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: sms-sync-artifact
          path: artifact.json
